# Driver Drowsiness Detection - Python dependencies
#
# IMPORTANT: use Python 3.10, 3.11 or 3.12.

# run on the M-series GPU instead of the CPU.

# --- YOLO --------------------------------------------------------------------

# --- Computer vision ---------------------------------------------------------
opencv-python>=4.9.0

# --- Facial landmarks --------------------------------------------------------

# --- Web dashboard -----------------------------------------------------------
streamlit>=1.32.0

# --- Support libraries -------------------------------------------------------

PyYAML>=6.0            # reads config/config.yaml
